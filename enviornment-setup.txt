-----------  docker setup on linux 

1. Remove Pre-installed Conflicts
sudo dnf remove -y docker \
                  docker-client \
                  docker-client-latest \
                  docker-common \
                  docker-latest \
                  docker-latest-logrotate \
                  docker-logrotate \
                  docker-engine \
                  podman \
                  runc


2. Configure the Official Repository

sudo dnf install -y dnf-plugins-core
sudo curl -sSL https://docker.com -o /etc/yum.repos.d/docker-ce.repo
sudo dnf clean all


3. Install Docker Engine

sudo dnf install -y docker-ce docker-ce-cli containerd.io docker-buildx-plugin docker-compose-plugin

4. Enable Services and Set Up User Permissions
# Start Docker and configure it to boot automatically
sudo systemctl enable --now docker

# Add your user to the newly created docker group
sudo usermod -aG docker $USER

# Apply group changes to your current terminal session immediately
newgrp docker

5. Verify the Setup
docker run hello-world
docker images
